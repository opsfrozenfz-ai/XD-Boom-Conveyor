ตั้งค่าระบบหลังบ้าน: Google Sheet (บันทึกตารางรายวัน) & Google Drive (เก็บ 2 ไฟล์)

เชื่อมต่อ Google Sheet สำหรับบันทึกตารางสรุปผลการเปรียบเทียบรายวันพร้อมหมายเหตุ และเก็บไฟล์ลง Google Drive

🌐 วิธีทำให้โค้ดฝังติดไปกับไฟล์ HTML และอัปขึ้นเว็บออนไลน์ให้ทุกคนดูได้:

  1. วางโค้ด Google Apps Script ด้านล่างใน Google Sheet ของคุณ แล้วกด Deploy → New deployment → Web app (เปิดสิทธิ์ Anyone / ทุกคน)
  2. นำลิงก์ Web App URL ที่ได้มาวางในช่องที่ 3 ด้านล่าง แล้วกดปุ่มสีเขียว "📦 ฝังลิงก์+ข้อมูลลงโค้ด & ดาวน์โหลด index.html สำหรับอัปขึ้นเว็บ"
  3. ไฟล์ index.html ที่ดาวน์โหลดออกมาจะมีลิงก์หลังบ้านและข้อมูลปัจจุบันฝังอยู่ในโค้ดถาวร สามารถนำไปอัปโหลดขึ้นเว็บฟรี เช่น Netlify Drop, GitHub Pages, Firebase หรือใส่ใน Google Apps Script เพื่อให้ทุกคนเปิดดูออนไลน์และเห็นข้อมูลชุดเดียวกันได้ทันที!

ลิงก์โฟลเดอร์ Google Drive ที่ใช้เก็บไฟล์ Booking และ Receiving History

วาง URL ที่ได้จากการ Deploy Google Apps Script ด้านล่าง (รองรับทั้งการบันทึกตารางรายวันลง Google Sheet และการอัปโหลดทั้ง 2 ไฟล์ลง Google Drive ในลิงก์เดียว)

โค้ด Google Apps Script ฝังหลังบ้าน (รองรับดึงข้อมูลออนไลน์ doGet + บันทึกข้อมูล doPost) วิธีใช้: เปิด Google Sheet ของคุณ → เมนู Extensions (ส่วนขยาย) → Apps Script → วางโค้ดนี้ → กด Deploy เป็น Web App (Anyone) 
/**
 * Google Apps Script หลังบ้านสำหรับ CDC Inbound Dashboard (รองรับออนไลน์ทุกคนดูได้)
 * - doGet(e): ส่งข้อมูลล่าสุดจาก Google Sheet ให้ทุกคนที่เปิดหน้าเว็บออนไลน์เห็นตรงกันอัตโนมัติ
 * - doPost(e): บันทึกตารางรายวัน + หมายเหตุ + ข้อมูลกราฟลง Google Sheet และเก็บไฟล์ลง Google Drive
 */
var FOLDER_ID = "ใส่_FOLDER_ID_ของ_GOOGLE_DRIVE_ที่นี่"; // เว้นว่างได้ ระบบจะเก็บไว้ที่ Root Drive
var SHEET_NAME = "Daily_Summary_Boom";
var STATE_SHEET_NAME = "Dashboard_Cloud_State";

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var stateSheet = ss.getSheetByName(STATE_SHEET_NAME);
    var sharedState = null;

    if (stateSheet && stateSheet.getLastRow() >= 2) {
      var rawJson = stateSheet.getRange(2, 1).getValue();
      if (rawJson) sharedState = JSON.parse(rawJson);
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: "ok",
      sharedState: sharedState
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var result = { status: "ok" };

    // 1) บันทึกตารางสรุปผลการเปรียบเทียบรายวัน + หมายเหตุ ลงชีต "Daily_Summary_Boom"
    if (payload.dailySummary && payload.dailySummary.length > 0) {
      var sheet = ss.getSheetByName(SHEET_NAME);
      if (!sheet) {
        sheet = ss.insertSheet(SHEET_NAME);
        sheet.appendRow([
          "วันที่ (Date)",
          "Plan 100% (Kg.)",
          "Target 80% Plan (Kg.)",
          "Actual Rec WT (Kg.)",
          "ส่วนต่างเทียบ Target 80% (Kg.)",
          "% Achievement",
          "สถานะ Target 80%",
          "หมายเหตุ / เหตุผลการปฏิบัติงาน",
          "วันเวลาที่บันทึก (Updated At)"
        ]);
        sheet.getRange(1, 1, 1, 9).setBackground("#C54B6C").setFontColor("#FFFFFF").setFontWeight("bold");
      }

      var existingData = sheet.getDataRange().getValues();
      var dateRowMap = {};
      for (var i = 1; i < existingData.length; i++) {
        var rowDate = String(existingData[i][0]).trim();
        if (rowDate) dateRowMap[rowDate] = i + 1;
      }

      var rows = payload.dailySummary || [];
      var timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss");

      rows.forEach(function(r) {
        var rowValues = [
          r.date,
          Number(r.planKg || 0),
          Number(r.target80Kg || 0),
          Number(r.actualKg || 0),
          Number(r.diffTarget80 || 0),
          Number(r.achievePct || 0).toFixed(1) + "%",
          r.targetStatus || "",
          r.remark || "",
          timestamp
        ];
        if (dateRowMap[r.date]) {
          sheet.getRange(dateRowMap[r.date], 1, 1, 9).setValues([rowValues]);
        } else {
          sheet.appendRow(rowValues);
        }
      });
      result.savedRows = rows.length;
    }

    // 2) บันทึกสถานะข้อมูลทั้งหมดไว้ในชีต "Dashboard_Cloud_State" เพื่อให้ทุกคนที่เปิดเว็บออนไลน์เห็นข้อมูลตรงกัน
    if (payload.sharedState) {
      var stateSheet = ss.getSheetByName(STATE_SHEET_NAME);
      if (!stateSheet) {
        stateSheet = ss.insertSheet(STATE_SHEET_NAME);
        stateSheet.appendRow(["JSON_STATE", "UPDATED_AT"]);
      }
      var nowStr = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss");
      stateSheet.getRange(2, 1, 1, 2).setValues([[JSON.stringify(payload.sharedState), nowStr]]);
    }

    // 3) บันทึกไฟล์ทั้ง 2 ไฟล์ (Booking & Receiving History) ลง Google Drive
    if (payload.files && payload.files.length > 0) {
      var folder = (FOLDER_ID && FOLDER_ID !== "ใส่_FOLDER_ID_ของ_GOOGLE_DRIVE_ที่นี่")
        ? DriveApp.getFolderById(FOLDER_ID)
        : DriveApp.getRootFolder();
      var savedFiles = [];
      payload.files.forEach(function(f) {
        var bytes = Utilities.base64Decode(f.base64);
        var blob = Utilities.newBlob(bytes, f.mimeType, f.fileName);
        var file = folder.createFile(blob);
        savedFiles.push({ name: file.getName(), url: file.getUrl() });
      });
      result.files = savedFiles;
    }

    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
